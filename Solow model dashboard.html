// Core simulation: mirrors the original R accumulate() logic.
// Each period's flows (Y, I, C) are computed from that period's stocks;
// stocks next period = this period's stock + this period's change.
// A shock (if enabled) switches ALL exogenous parameters (alpha, s, n, delta)
// to their new values from shockTime onward. Initial stocks K0/L0 are not shocked.
simulate = {
  const results = [];
  let L = L0, K = K0;
  for (let t = 0; t <= Tmax; t++) {
    const shocked = shockOn && t >= shockTime;
    const aC = shocked ? alphaNew : alpha;
    const sC = shocked ? sNew : s;
    const nC = shocked ? nNew : n;
    const dC = shocked ? deltaNew : delta;

    const Y = Math.pow(K, aC) * Math.pow(L, 1 - aC);
    const I = sC * Y;
    const C = (1 - sC) * Y;
    const k = K / L;
    const y = Y / L;
    results.push({t, L, K, Y, I, C, k, y, s: sC, alpha: aC, n: nC, delta: dC});

    const L_dot = nC * L;
    const K_dot = sC * Y - dC * K;
    L = L + L_dot;
    K = K + K_dot;
  }
  return results;
}
dataWithGrowth = simulate.map((d, i, arr) => ({
  ...d,
  growth: i === 0 ? null : (d.Y / arr[i - 1].Y - 1)
}))
variableLabels = new Map([
  ["k", "Capital-labor ratio (k)"],
  ["y", "Output per worker (y)"],
  ["Y", "Output (Y)"],
  ["K", "Capital (K)"],
  ["L", "Labor (L)"],
  ["C", "Consumption (C)"],
  ["I", "Investment (I)"],
  ["growth", "Output growth rate"]
])
kStar = Math.pow(s / (n + delta), 1 / (1 - alpha))
kStarNew = Math.pow(sNew / (nNew + deltaNew), 1 / (1 - alphaNew))
kMaxDiagram = Math.max(kStar, shockOn ? kStarNew : 0, d3.max(dataWithGrowth, d => d.k)) * 1.3
solowData = d3.range(0, kMaxDiagram, kMaxDiagram / 200).map(k => ({
  k,
  sf_k: s * Math.pow(k, alpha),
  break_even: (n + delta) * k,
  sf_k_new: sNew * Math.pow(k, alphaNew),
  break_even_new: (nNew + deltaNew) * k
}))
currentPoint = {
  const last = dataWithGrowth[dataWithGrowth.length - 1];
  return {k: last.k, sf_k: last.s * Math.pow(last.k, last.alpha)};
}
Plot.plot({
  title: "Transition Path",
  width: 760,
  height: 380,
  grid: true,
  x: {label: "Time (t)"},
  y: {label: variableLabels.get(variable)},
  marks: [
    Plot.lineY(dataWithGrowth, {x: "t", y: variable, stroke: "#2563eb", strokeWidth: 2}),
    shockOn ? Plot.ruleX([shockTime], {stroke: "#dc2626", strokeDasharray: "4,3"}) : null,
    Plot.ruleY([0])
  ]
})
Plot.plot({
  title: "Solow Diagram",
  width: 760,
  height: 380,
  grid: true,
  x: {label: "Capital per worker (k)", domain: [0, kMaxDiagram]},
  y: {label: "Output / Investment per worker"},
  marks: [
    Plot.lineY(solowData, {x: "k", y: "sf_k", stroke: "#2563eb", strokeWidth: 2}),
    Plot.lineY(solowData, {x: "k", y: "break_even", stroke: "#111827", strokeWidth: 2}),
    shockOn ? Plot.lineY(solowData, {x: "k", y: "sf_k_new", stroke: "#2563eb", strokeWidth: 2, strokeDasharray: "5,3"}) : null,
    shockOn ? Plot.lineY(solowData, {x: "k", y: "break_even_new", stroke: "#111827", strokeWidth: 2, strokeDasharray: "5,3"}) : null,
    Plot.ruleX([kStar], {stroke: "#94a3b8", strokeDasharray: "2,2"}),
    shockOn ? Plot.ruleX([kStarNew], {stroke: "#94a3b8", strokeDasharray: "2,2"}) : null,
    Plot.dot([currentPoint], {x: "k", y: "sf_k", r: 5, fill: "#dc2626"})
  ]
})
